# Makefile para BaraOS

CC = gcc
AS = nasm
CFLAGS = -m32 -nostdlib -nostdinc -fno-builtin -fno-stack-protector -nostartfiles -nodefaultlibs -Wall -Wextra -Werror -c
LDFLAGS = -m elf_i386 -T link.ld
ASFLAGS = -f elf

SOURCES_C = kernel.c screen.c keyboard.c filesystem.c string.c memory.c
SOURCES_ASM = boot.asm
OBJECTS_C = $(SOURCES_C:.c=.o)
OBJECTS_ASM = boot.o

all: os-image.bin

os-image.bin: boot.bin kernel.bin
	cat boot.bin kernel.bin > os-image.bin

boot.bin: boot.asm
	nasm -f bin boot.asm -o boot.bin

kernel.bin: kernel.elf
	objcopy -O binary kernel.elf kernel.bin

kernel.elf: $(OBJECTS_C) kernel_entry.o
	ld $(LDFLAGS) -o kernel.elf kernel_entry.o $(OBJECTS_C)

kernel_entry.o: kernel_entry.asm
	nasm -f elf kernel_entry.asm -o kernel_entry.o

%.o: %.c
	$(CC) $(CFLAGS) $< -o $@

%.o: %.asm
	$(AS) $(ASFLAGS) $< -o $@

clean:
	rm -rf *.bin *.elf *.o os-image.bin

run: os-image.bin
	qemu-system-i386 -fda os-image.bin

.PHONY: clean run
